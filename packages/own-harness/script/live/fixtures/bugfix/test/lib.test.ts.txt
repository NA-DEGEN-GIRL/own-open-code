import { describe, expect, test } from "bun:test"
import { parseCsv } from "../src/csv"
import { addBusinessDays } from "../src/dates"
import { LRU } from "../src/lru"

describe("parseCsv", () => {
  test("plain rows", () => expect(parseCsv("a,b\nc,d")).toEqual([["a", "b"], ["c", "d"]]))
  test("quoted comma", () => expect(parseCsv('a,"b,c"')).toEqual([["a", "b,c"]]))
  test("doubled quotes inside a quoted field", () => expect(parseCsv('"he said ""hi"""')).toEqual([['he said "hi"']]))
  test("CRLF rows", () => expect(parseCsv("a,b\r\nc,d")).toEqual([["a", "b"], ["c", "d"]]))
  test("a trailing newline adds no row", () => expect(parseCsv("a,b\n")).toEqual([["a", "b"]]))
  test("empty fields", () => expect(parseCsv("a,,c")).toEqual([["a", "", "c"]]))
  test("empty input", () => expect(parseCsv("")).toEqual([]))
  test("newline inside quotes", () => expect(parseCsv('"x\ny",z')).toEqual([["x\ny", "z"]]))
})

describe("addBusinessDays", () => {
  test("Wednesday + 2 is Friday", () => expect(addBusinessDays("2026-10-14", 2)).toBe("2026-10-16"))
  test("Friday + 1 is Monday", () => expect(addBusinessDays("2026-10-16", 1)).toBe("2026-10-19"))
  test("Saturday + 1 is Monday", () => expect(addBusinessDays("2026-10-17", 1)).toBe("2026-10-19"))
  test("Friday + 5 is the next Friday", () => expect(addBusinessDays("2026-10-16", 5)).toBe("2026-10-23"))
  test("zero days is the same date", () => expect(addBusinessDays("2026-10-14", 0)).toBe("2026-10-14"))
  test("a negative count is a RangeError", () => expect(() => addBusinessDays("2026-10-14", -1)).toThrow(RangeError))
})

describe("LRU", () => {
  test("reading a key keeps it", () => {
    const cache = new LRU<string, number>(2)
    cache.set("a", 1)
    cache.set("b", 2)
    expect(cache.get("a")).toBe(1)
    cache.set("c", 3)
    expect(cache.get("b")).toBeUndefined()
    expect(cache.get("a")).toBe(1)
    expect(cache.get("c")).toBe(3)
  })
  test("updating a key keeps it", () => {
    const cache = new LRU<string, number>(2)
    cache.set("a", 1)
    cache.set("b", 2)
    cache.set("a", 10)
    cache.set("c", 3)
    expect(cache.get("b")).toBeUndefined()
    expect(cache.get("a")).toBe(10)
    expect(cache.size).toBe(2)
  })
})
